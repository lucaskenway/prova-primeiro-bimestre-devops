## 3. CRUD

## 4. Validações
$ curl -s -w ' [HTTP %{http_code}]' -X POST localhost:3001/reservas -H 'Content-Type: application/json' -d '{"data":"2026-10-01T14:00:00Z"}'  # sem cliente
{"erro":"Campos obrigatórios: cliente, data"} [HTTP 400]

$ curl -s -w ' [HTTP %{http_code}]' -X POST localhost:3001/reservas -H 'Content-Type: application/json' -d '{"cliente":"Ana"}'  # sem data
{"erro":"Campos obrigatórios: cliente, data"} [HTTP 400]

$ curl -s -w ' [HTTP %{http_code}]' -X POST localhost:3001/reservas -H 'Content-Type: application/json' -d '{"cliente":"Ana","data":"2026-10-01T14:00:00Z","status":"aprovada"}'  # status inválido
{"erro":"status deve ser um de: pendente, confirmada, cancelada"} [HTTP 400]

$ curl -s -w ' [HTTP %{http_code}]' -X POST localhost:3001/reservas -H 'Content-Type: application/json' -d '{"cliente":"Ana","data":"01/10/2026"}'  # não ISO 8601
{"erro":"data inválida (use ISO 8601, ex: 2026-10-01T14:00:00Z)"} [HTTP 400]

$ curl -s -w ' [HTTP %{http_code}]' -X POST localhost:3001/reservas -H 'Content-Type: application/json' -d '{"cliente":"Ana","data":"October 1, 2026"}'  # não ISO 8601
{"erro":"data inválida (use ISO 8601, ex: 2026-10-01T14:00:00Z)"} [HTTP 400]

$ curl -s -w ' [HTTP %{http_code}]' -X POST localhost:3001/reservas -H 'Content-Type: application/json' -d '{"cliente":"Ana","data":"2026-02-30"}'  # dia inexistente
{"erro":"data inválida (use ISO 8601, ex: 2026-10-01T14:00:00Z)"} [HTTP 400]

$ curl -s -w ' [HTTP %{http_code}]' -X POST localhost:3001/reservas -H 'Content-Type: application/json' -d '{"cliente":"Ana","data":123}'  # não é string
{"erro":"data inválida (use ISO 8601, ex: 2026-10-01T14:00:00Z)"} [HTTP 400]

## 5. /health com o banco indisponível
$ curl -s -o /dev/null localhost:3001/reservas  # deixa conexão ociosa no pool


$ docker compose stop db 2>&1
 Container reservas-db Stopping 
 Container reservas-db Stopped 


$ curl -s -m 10 -w ' [HTTP %{http_code}]' localhost:3001/health
{"status":"erro","db":"indisponivel"} [HTTP 503]

$ curl -s -m 10 -w ' [HTTP %{http_code}]' localhost:3001/reservas
{"erro":"Erro interno"} [HTTP 500]

$ docker inspect reservas-api --format 'api: {{.State.Status}} RestartCount={{.RestartCount}}'
api: running RestartCount=0


$ docker compose start db 2>&1
 Container reservas-db Starting 
 Container reservas-db Started 


$ curl -s -w ' [HTTP %{http_code}]' localhost:3001/health  # recuperou sem reiniciar a API
{"status":"ok","db":"ok"} [HTTP 200]
